#include <tunables/global>

profile acoustica flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Add-on data
  /data/ rw,
  /data/** rw,

  # Python and application
  /usr/bin/python3* ix,
  # Python console scripts: the interpreter READS the script file, so 'r' is
  # required ('/usr/bin/** ix' only grants execute). Lets `acoustic-engine ...`
  # run in the container (e.g. for debugging).
  /usr/bin/acoustic-engine rix,
  /usr/bin/alarm-tuner rix,
  /usr/lib/python3*/** r,
  /app/ r,           # read (list) the dir itself — needed so `python -m` can find the package
  /app/** r,
  /app/run.sh rix,

  # Audio access (PulseAudio)
  /run/audio/pulse.sock rw,
  /root/.config/pulse/** rw,

  # Network for WebSocket to Home Assistant
  network inet stream,
  network inet6 stream,

  # Config directory access (for auto-installing integration)
  /config/** rw,
  /homeassistant/** rw,

  # System libraries
  /lib/** rm,
  /usr/lib/** rm,
  # ALSA runtime config (defines the 'pulse' plugin for PortAudio)
  /usr/share/alsa/ r,
  /usr/share/alsa/** r,
  # HA injects /etc/asound.conf (the pulse default). ALSA's config hook loads it,
  # and if it CAN'T be read it discards the ENTIRE config -> all audio fails with
  # "hooks failed, removing configuration". Must be readable.
  /etc/asound.conf r,
  /etc/alsa/ r,
  /etc/alsa/** r,
  /etc/pulse/** r,

  # Device access for audio
  /dev/snd/* rw,
  /dev/audio* rw,
  # PulseAudio client shared memory — the pulse plugin/PortAudio fail to init
  # without this (audio is moved over POSIX shm).
  /dev/shm/ rw,
  /dev/shm/** rwk,

  # Proc and sys for audio detection + numpy/openblas CPU probing
  /proc/*/status r,
  /proc/asound/** r,
  /proc/cpuinfo r,
  /proc/meminfo r,
  /sys/devices/** r,

  # DNS resolution
  /etc/nsswitch.conf r,
  /etc/hosts r,
  /etc/resolv.conf r,
  /etc/host.conf r,
  /etc/gai.conf r,

  # Temporary files
  /tmp/** rw,
  /var/tmp/** rw,
}
